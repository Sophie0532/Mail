import os
import pandas as pd
import gzip

def procesar_cftc_especifico():
    ruta_base = r"X:\REGULATORIO MERCADOS\FICHEROS DTCC PORTAL EU\DFA\2025\10. Octubre"
    
    todos_datos = []
    
    # Para cada día del mes
    for dia in range(1, 32):
        carpeta = f"{dia:02d}.10"
        ruta_completa = os.path.join(ruta_base, carpeta)
        
        if os.path.exists(ruta_completa):
            print(f"Revisando {carpeta}...")
            
            # Listar todos los archivos en la carpeta
            for archivo in os.listdir(ruta_completa):
                ruta_archivo = os.path.join(ruta_completa, archivo)
                
                # Verificar patrones específicos
                if (archivo.startswith("CFTC_IR_SubmissionReport_") and archivo.endswith("-en-6STB.csv.gz") or
                    archivo.startswith("CFTC_FX_SubmissionReport_") and archivo.endswith("-en-5STB.csv.gz") or
                    archivo.startswith("CFTC_EQ_SubmissionReport_") and archivo.endswith("-en-4STB.csv.gz") or
                    archivo.startswith("CFTC_CR_SubmissionReport_") and archivo.endswith("-en-3STB.csv.gz")):
                    
                    print(f"  → Encontrado: {archivo}")
                    
                    try:
                        with gzip.open(ruta_archivo, 'rt', encoding='utf-8') as f:
                            df = pd.read_csv(f)
                        
                        df['Origen'] = archivo
                        df['Dia'] = f"2025-10-{dia:02d}"
                        todos_datos.append(df)
                        
                    except Exception as e:
                        print(f"    Error en {archivo}: {e}")
    
    # Guardar resultado
    if todos_datos:
        df_final = pd.concat(todos_datos, ignore_index=True)
        df_final.to_excel("Reportes_CFTC_Combinados.xlsx", index=False)
        print(f"✓ Excel creado con {len(todos_datos)} archivos y {len(df_final)} registros")

# Ejecutar
procesar_cftc_especifico()
