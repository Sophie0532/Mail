import os
import gzip
import pandas as pd
from pathlib import Path
import glob
import csv

def procesar_archivos_mes_octubre_con_consolidado():
    # Configuración de rutas
    ruta_base = r"\\v2sevi3.san.corp\Factory\REGULATORIO MERCADOS\FICHEROS DTCC PORTAL EU\DFA\2025\10. Octubre"
    archivo_salida = r"\\v2sevi3.san.corp\Factory\REGULATORIO MERCADOS\FICHEROS DTCC PORTAL EU\DFA\2025\Consolidado_Octubre_2025.xlsx"
    
    # Patrón del archivo a buscar
    patron_archivo = "CFTC_IR_SubmissionReport_*-en-6STB.csv.gz"
    
    # Lista para controlar si hay datos para guardar
    hojas_creadas = False
    todos_los_dataframes = []  # Lista para almacenar todos los DataFrames consolidados
    
    # Crear un escritor de Excel
    with pd.ExcelWriter(archivo_salida, engine='openpyxl') as writer:
        
        # Procesar cada día del mes (01 al 31)
        for dia in range(1, 32):
            # Formatear día con dos dígitos
            dia_formateado = f"{dia:02d}"
            carpeta_dia = os.path.join(ruta_base, f"{dia_formateado}.10")
            
            # Verificar si existe la carpeta del día
            if not os.path.exists(carpeta_dia):
                print(f"Carpeta no encontrada: {carpeta_dia}")
                continue
            
            print(f"Procesando día: {dia_formateado}.10")
            
            # Buscar archivos que coincidan con el patrón
            patron_busqueda = os.path.join(carpeta_dia, patron_archivo)
            archivos = glob.glob(patron_busqueda)
            
            if not archivos:
                print(f"  No se encontraron archivos con el patrón en {dia_formateado}.10")
                continue
            
            # Lista para almacenar todos los DataFrames del día
            dataframes_dia = []
            
            # Procesar cada archivo del día
            for archivo in archivos:
                nombre_archivo = os.path.basename(archivo)
                print(f"  Procesando: {nombre_archivo}")
                
                try:
                    # Método 1: Intentar con pandas primero
                    with gzip.open(archivo, 'rt', encoding='utf-8') as f:
                        df = pd.read_csv(f)
                    
                    if not df.empty:
                        dataframes_dia.append(df)
                        
                except pd.errors.ParserError:
                    # Método 2: Si falla pandas, usar el módulo csv con manejo de errores
                    try:
                        print(f"    Usando método alternativo para: {nombre_archivo}")
                        filas = []
                        with gzip.open(archivo, 'rt', encoding='utf-8') as f:
                            lector = csv.reader(f, delimiter=',', quotechar='"', quoting=csv.QUOTE_MINIMAL)
                            headers = next(lector)  # Leer headers
                            for fila in lector:
                                filas.append(fila)
                        
                        if filas:
                            df = pd.DataFrame(filas, columns=headers)
                            dataframes_dia.append(df)
                            
                    except Exception as e:
                        print(f"    Error con método CSV: {str(e)}")
                        # Método 3: Leer como texto y limpiar
                        try:
                            print(f"    Usando método de limpieza de texto para: {nombre_archivo}")
                            with gzip.open(archivo, 'rt', encoding='utf-8') as f:
                                lineas = f.readlines()
                            
                            if len(lineas) > 1:
                                # Asumir que la primera línea son los headers
                                headers = [h.strip().strip('"') for h in lineas[0].strip().split(',')]
                                datos = []
                                
                                for linea in lineas[1:]:
                                    campos = [c.strip().strip('"') for c in linea.strip().split(',')]
                                    if len(campos) == len(headers):
                                        datos.append(campos)
                                    else:
                                        # Si no coinciden, intentar unir los campos extras al último campo
                                        if len(campos) > len(headers):
                                            campos_corregidos = campos[:len(headers)-1] + [','.join(campos[len(headers)-1:])]
                                            datos.append(campos_corregidos)
                                
                                if datos:
                                    df = pd.DataFrame(datos, columns=headers)
                                    dataframes_dia.append(df)
                                    
                        except Exception as e3:
                            print(f"    Error crítico procesando {nombre_archivo}: {str(e3)}")
                            continue
                
                except Exception as e:
                    print(f"    Error general al procesar {nombre_archivo}: {str(e)}")
                    continue
            
            # Consolidar todos los DataFrames del día
            if dataframes_dia:
                try:
                    # Concatenar todos los DataFrames del día
                    df_consolidado_dia = pd.concat(dataframes_dia, ignore_index=True)
                    
                    # Agregar columna con la fecha del día
                    df_consolidado_dia['Fecha_Procesamiento'] = f"2025-10-{dia_formateado}"
                    
                    # Agregar a la lista general para el consolidado final
                    todos_los_dataframes.append(df_consolidado_dia)
                    
                    # Nombre de la hoja (máximo 31 caracteres para Excel)
                    nombre_hoja = f"Dia_{dia_formateado}"
                    if len(nombre_hoja) > 31:
                        nombre_hoja = nombre_hoja[:31]
                    
                    # Guardar en Excel (sin la columna adicional para las hojas individuales)
                    df_consolidado_dia.drop('Fecha_Procesamiento', axis=1, inplace=True)
                    df_consolidado_dia.to_excel(writer, sheet_name=nombre_hoja, index=False)
                    hojas_creadas = True
                    
                    print(f"  Día {dia_formateado}.10 completado - {len(archivos)} archivos procesados, {len(df_consolidado_dia)} registros")
                    
                except Exception as e:
                    print(f"  Error al consolidar día {dia_formateado}.10: {str(e)}")
            else:
                print(f"  No se pudieron procesar archivos para el día {dia_formateado}.10")
        
        # Crear hoja consolidada con todos los días
        if todos_los_dataframes:
            try:
                print("\nCreando hoja consolidada con todos los días...")
                df_consolidado_total = pd.concat(todos_los_dataframes, ignore_index=True)
                
                # Guardar hoja consolidada
                df_consolidado_total.to_excel(writer, sheet_name='CONSOLIDADO_TOTAL', index=False)
                print(f"  Hoja consolidada creada con {len(df_consolidado_total)} registros totales")
                
            except Exception as e:
                print(f"  Error al crear hoja consolidada: {str(e)}")
        
        # Si no se crearon hojas, crear una hoja vacía para evitar el error
        if not hojas_creadas:
            df_vacio = pd.DataFrame({'Mensaje': ['No se encontraron datos procesables en ningún archivo']})
            df_vacio.to_excel(writer, sheet_name='Sin_Datos', index=False)
            print("Se creó hoja 'Sin_Datos' porque no se encontraron datos procesables")
    
    print(f"\nProceso completado. Archivo guardado en: {archivo_salida}")

# Versión alternativa que también incluye estadísticas
def procesar_archivos_con_consolidado_y_estadisticas():
    # Configuración de rutas
    ruta_base = r"\\v2sevi3.san.corp\Factory\REGULATORIO MERCADOS\FICHEROS DTCC PORTAL EU\DFA\2025\10. Octubre"
    archivo_salida = r"\\v2sevi3.san.corp\Factory\REGULATORIO MERCADOS\FICHEROS DTCC PORTAL EU\DFA\2025\Consolidado_Octubre_2025_Completo.xlsx"
    
    # Patrón del archivo a buscar
    patron_archivo = "CFTC_IR_SubmissionReport_*-en-6STB.csv.gz"
    
    todos_los_dataframes = []
    estadisticas = []
    
    with pd.ExcelWriter(archivo_salida, engine='openpyxl') as writer:
        
        for dia in range(1, 32):
            dia_formateado = f"{dia:02d}"
            carpeta_dia = os.path.join(ruta_base, f"{dia_formateado}.10")
            
            if not os.path.exists(carpeta_dia):
                continue
            
            print(f"Procesando día: {dia_formateado}.10")
            
            patron_busqueda = os.path.join(carpeta_dia, patron_archivo)
            archivos = glob.glob(patron_busqueda)
            
            if not archivos:
                continue
            
            dataframes_dia = []
            
            for archivo in archivos:
                nombre_archivo = os.path.basename(archivo)
                
                try:
                    with gzip.open(archivo, 'rt', encoding='utf-8') as f:
                        df = pd.read_csv(f)
                    
                    if not df.empty:
                        dataframes_dia.append(df)
                        
                except pd.errors.ParserError:
                    try:
                        with gzip.open(archivo, 'rt', encoding='utf-8') as f:
                            lector = csv.reader(f, delimiter=',', quotechar='"', quoting=csv.QUOTE_MINIMAL)
                            headers = next(lector)
                            filas = [fila for fila in lector]
                        
                        if filas:
                            df = pd.DataFrame(filas, columns=headers)
                            dataframes_dia.append(df)
                            
                    except Exception:
                        continue
            
            if dataframes_dia:
                try:
                    df_consolidado_dia = pd.concat(dataframes_dia, ignore_index=True)
                    
                    # Agregar columna con la fecha
                    df_consolidado_dia['Fecha_Dia'] = f"2025-10-{dia_formateado}"
                    
                    # Guardar estadísticas
                    estadisticas.append({
                        'Dia': f"2025-10-{dia_formateado}",
                        'Archivos_Procesados': len(archivos),
                        'Registros_Totales': len(df_consolidado_dia),
                        'Columnas': len(df_consolidado_dia.columns)
                    })
                    
                    # Agregar a la lista general
                    todos_los_dataframes.append(df_consolidado_dia)
                    
                    # Guardar hoja individual
                    nombre_hoja = f"Dia_{dia_formateado}"
                    df_sin_fecha = df_consolidado_dia.drop('Fecha_Dia', axis=1)
                    df_sin_fecha.to_excel(writer, sheet_name=nombre_hoja, index=False)
                    
                    print(f"  Día {dia_formateado}.10: {len(archivos)} archivos, {len(df_consolidado_dia)} registros")
                    
                except Exception as e:
                    print(f"  Error día {dia_formateado}.10: {str(e)}")
        
        # Hoja consolidada total
        if todos_los_dataframes:
            df_consolidado_total = pd.concat(todos_los_dataframes, ignore_index=True)
            df_consolidado_total.to_excel(writer, sheet_name='CONSOLIDADO_TOTAL', index=False)
            print(f"\nCONSOLIDADO TOTAL: {len(df_consolidado_total)} registros")
        
        # Hoja de estadísticas
        if estadisticas:
            df_estadisticas = pd.DataFrame(estadisticas)
            df_estadisticas.to_excel(writer, sheet_name='ESTADISTICAS', index=False)
            print("Hoja de ESTADISTICAS creada")
        
        # Resumen ejecutivo
        if todos_los_dataframes:
            resumen = {
                'Metrica': [
                    'Total de días procesados',
                    'Total de registros consolidados',
                    'Total de columnas',
                    'Fecha inicio',
                    'Fecha fin',
                    'Promedio registros por día'
                ],
                'Valor': [
                    len(estadisticas),
                    len(df_consolidado_total) if todos_los_dataframes else 0,
                    len(df_consolidado_total.columns) if todos_los_dataframes else 0,
                    min([e['Dia'] for e in estadisticas]) if estadisticas else 'N/A',
                    max([e['Dia'] for e in estadisticas]) if estadisticas else 'N/A',
                    sum([e['Registros_Totales'] for e in estadisticas]) / len(estadisticas) if estadisticas else 0
                ]
            }
            df_resumen = pd.DataFrame(resumen)
            df_resumen.to_excel(writer, sheet_name='RESUMEN', index=False)
            print("Hoja de RESUMEN creada")
    
    print(f"\nArchivo completo guardado en: {archivo_salida}")
