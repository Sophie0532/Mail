import pandas as pd
import os

# VERSIÓN SIMPLE Y DIRECTA
def consolidar_simple():
    # Ruta exacta de tu archivo
    archivo = r"\\v2sevi3.san.corp\Factory\REGULATORIO MERCADOS\FICHEROS DTCC PORTAL EU\DFA\2025\10. Octubre\Consolidado_OctubreCR_2025.xlsx"
    
    # Leer todas las hojas
    hojas = pd.read_excel(archivo, sheet_name=None)
    
    # Agregar nombre de hoja a cada DataFrame y consolidar
    datos_consolidados = []
    
    for nombre_hoja, datos in hojas.items():
        datos['Dia'] = nombre_hoja  # Agregar columna con el día
        datos_consolidados.append(datos)
    
    # Unir todos los datos
    df_final = pd.concat(datos_consolidados, ignore_index=True)
    
    # Guardar resultado
    df_final.to_excel(archivo.replace('.xlsx', '_CONSOLIDADO.xlsx'), index=False)
    
    print(f"✅ Listo! {len(hojas)} hojas consolidadas en un archivo")
    print(f"📊 Total de registros: {len(df_final)}")

# Ejecutar la versión simple
consolidar_simple()
