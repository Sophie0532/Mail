import os
import pandas as pd
import glob
from pathlib import Path

def procesar_archivos_mensuales():
    # Ruta base donde están los archivos
    ruta_base = r"\\v2sevi3.san.corp\Factory\REGULATORIO MERCADOS\FICHEROS DTCC PORTAL EU\DFA\2025\10. Octubre"
    
    # Patrones de archivos a buscar
    patrones = {
        'CFTC_IR': 'CFTC_IR_SubmissionReport_*-en-6STB.csv.gz',
        'CFTC_FX': 'CFTC_FX_SubmissionReport_*-en-5STB.csv.gz', 
        'CFTC_EQ': 'CFTC_EQ_SubmissionReport_*-en-4STB.csv.gz',
        'CFTC_CR': 'CFTC_CR_SubmissionReport_*-en-3STB.csv.gz'
    }
    
    # Diccionario para almacenar los DataFrames de cada tipo
    dataframes = {tipo: pd.DataFrame() for tipo in patrones.keys()}
    
    # Iterar por cada día del mes (01.10 a 31.10)
    for dia in range(1, 32):
        # Formatear el día con dos dígitos
        carpeta_dia = f"{dia:02d}.10"
        ruta_dia = os.path.join(ruta_base, carpeta_dia)
        
        # Verificar si la carpeta existe
        if not os.path.exists(ruta_dia):
            print(f"La carpeta {ruta_dia} no existe, saltando...")
            continue
            
        print(f"Procesando carpeta: {carpeta_dia}")
        
        # Procesar cada tipo de archivo
        for tipo, patron in patrones.items():
            # Buscar archivos que coincidan con el patrón
            patron_busqueda = os.path.join(ruta_dia, patron)
            archivos = glob.glob(patron_busqueda)
            
            if archivos:
                print(f"  Encontrados {len(archivos)} archivos de tipo {tipo}")
                
                # Leer y concatenar todos los archivos de este tipo para el día
                for archivo in archivos:
                    try:
                        # Leer el archivo comprimido
                        df_temp = pd.read_csv(archivo, compression='gzip')
                        
                        # Agregar columna con la fecha del archivo para trazabilidad
                        df_temp['Fecha_Archivo'] = f"2025-10-{dia:02d}"
                        
                        # Concatenar con el DataFrame principal de este tipo
                        if dataframes[tipo].empty:
                            dataframes[tipo] = df_temp
                        else:
                            dataframes[tipo] = pd.concat([dataframes[tipo], df_temp], ignore_index=True)
                            
                        print(f"    Procesado: {os.path.basename(archivo)}")
                        
                    except Exception as e:
                        print(f"    Error al procesar {archivo}: {e}")
            else:
                print(f"  No se encontraron archivos para {tipo} en {carpeta_dia}")
    
    # Guardar los archivos consolidados
    ruta_salida = ruta_base  # Puedes cambiar esto si quieres una carpeta diferente
    
    for tipo, df in dataframes.items():
        if not df.empty:
            # Nombre del archivo de salida
            nombre_archivo = f"CONSOLIDADO_{tipo}_OCTUBRE_2025.csv"
            ruta_completa = os.path.join(ruta_salida, nombre_archivo)
            
            # Guardar como CSV
            df.to_csv(ruta_completa, index=False)
            print(f"Archivo guardado: {ruta_completa}")
            
            # Opcional: guardar también como Excel
            # nombre_excel = f"CONSOLIDADO_{tipo}_OCTUBRE_2025.xlsx"
            # ruta_excel = os.path.join(ruta_salida, nombre_excel)
            # df.to_excel(ruta_excel, index=False)
            # print(f"Archivo Excel guardado: {ruta_excel}")
        else:
            print(f"No hay datos para {tipo}")
    
    print("Proceso completado!")

# Versión alternativa si prefieres especificar la ruta exacta
def procesar_archivos_mensuales_v2():
    ruta_base = r"\\v2sevi3.san.corp\Factory\REGULATORIO MERCADOS\FICHEROS DTCC PORTAL EU\DFA\2025\10. Octubre"
    
    # Lista de patrones exactos como los mencionaste
    patrones = [
        'CFTC_IR_SubmissionReport_*-en-6STB.csv.gz',
        'CFTC_FX_SubmissionReport_*-en-5STB.csv.gz',
        'CFTC_EQ_SubmissionReport_*-en-4STB.csv.gz', 
        'CFTC_CR_SubmissionReport_*-en-3STB.csv.gz'
    ]
    
    nombres_salida = [
        'CONSOLIDADO_CFTC_IR_OCTUBRE_2025.csv',
        'CONSOLIDADO_CFTC_FX_OCTUBRE_2025.csv',
        'CONSOLIDADO_CFTC_EQ_OCTUBRE_2025.csv',
        'CONSOLIDADO_CFTC_CR_OCTUBRE_2025.csv'
    ]
    
    # Procesar cada patrón
    for patron, nombre_salida in zip(patrones, nombres_salida):
        print(f"\nProcesando patrón: {patron}")
        
        # Lista para almacenar todos los DataFrames de este patrón
        todos_dfs = []
        
        # Buscar en todas las carpetas de días
        for dia in range(1, 32):
            carpeta_dia = f"{dia:02d}.10"
            ruta_busqueda = os.path.join(ruta_base, carpeta_dia, patron)
            archivos = glob.glob(ruta_busqueda)
            
            for archivo in archivos:
                try:
                    df = pd.read_csv(archivo, compression='gzip')
                    df['Fecha_Archivo'] = f"2025-10-{dia:02d}"  # Agregar fecha
                    todos_dfs.append(df)
                    print(f"  Procesado: {os.path.basename(archivo)}")
                except Exception as e:
                    print(f"  Error en {archivo}: {e}")
        
        # Consolidar y guardar
        if todos_dfs:
            df_consolidado = pd.concat(todos_dfs, ignore_index=True)
            ruta_salida = os.path.join(ruta_base, nombre_salida)
            df_consolidado.to_csv(ruta_salida, index=False)
            print(f"✓ Archivo consolidado guardado: {nombre_salida}")
            print(f"  Total de registros: {len(df_consolidado)}")
        else:
            print(f"✗ No se encontraron archivos para el patrón: {patron}")

if __name__ == "__main__":
    # Ejecutar el proceso
    procesar_archivos_mensuales_v2()
