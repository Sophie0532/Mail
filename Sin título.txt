import pandas as pd
import os
from datetime import datetime

def diagnostico_archivo_cftc(ruta_directorio, nombre_archivo):
    """
    Función de diagnóstico para entender la estructura real del archivo
    """
    ruta_completa = os.path.join(ruta_directorio, nombre_archivo)
    
    print("🔍 INICIANDO DIAGNÓSTICO DEL ARCHIVO")
    print("=" * 60)
    
    if not os.path.exists(ruta_completa):
        print(f"❌ El archivo no existe: {ruta_completa}")
        return
    
    try:
        # Leer el archivo Excel
        xls = pd.ExcelFile(ruta_completa)
        
        print(f"📊 Hojas encontradas: {len(xls.sheet_names)}")
        print(f"📋 Nombres de hojas: {xls.sheet_names}")
        
        # Analizar cada hoja que comience con "Dia_"
        for hoja in xls.sheet_names:
            if hoja.startswith('Dia_'):
                print(f"\n{'='*50}")
                print(f"📖 ANALIZANDO HOJA: {hoja}")
                print(f"{'='*50}")
                
                # Estrategia 1: Leer sin saltar filas para ver estructura real
                try:
                    df_raw = pd.read_excel(xls, sheet_name=hoja, header=None)
                    print(f"📐 Dimensiones: {df_raw.shape} (filas x columnas)")
                    
                    if df_raw.empty:
                        print("❌ Hoja completamente vacía")
                        continue
                    
                    print(f"📝 Primeras 5 filas RAW:")
                    for i in range(min(5, len(df_raw))):
                        fila = df_raw.iloc[i].dropna()  # Eliminar valores NaN
                        if len(fila) > 0:
                            print(f"   Fila {i}: {list(fila.values)}")
                        else:
                            print(f"   Fila {i}: [VACÍA]")
                    
                    # Buscar patrones de datos
                    print(f"\n🔎 Buscando patrones de datos...")
                    
                    # Buscar encabezados comunes
                    encabezados_comunes = ['Trade ID', 'Message Type', 'GTR Validation Status', 'Report Date']
                    for i in range(min(10, len(df_raw))):
                        fila = df_raw.iloc[i]
                        for j, cell in enumerate(fila):
                            if pd.notna(cell) and any(header in str(cell) for header in encabezados_comunes):
                                print(f"   ✅ Posible encabezado en fila {i}, columna {j}: '{cell}'")
                    
                    # Verificar si hay datos después de ciertas filas
                    for skip_rows in [0, 1, 2, 3, 4, 5]:
                        try:
                            df_test = pd.read_excel(xls, sheet_name=hoja, skiprows=skip_rows)
                            if not df_test.empty and len(df_test.columns) > 1:
                                print(f"   ✅ Con skiprows={skip_rows}: {len(df_test)} filas, {len(df_test.columns)} columnas")
                                # Mostrar nombres de columnas
                                print(f"      Columnas: {list(df_test.columns)}")
                        except Exception as e:
                            print(f"   ❌ Con skiprows={skip_rows}: Error - {str(e)}")
                            
                except Exception as e:
                    print(f"❌ Error leyendo hoja {hoja}: {str(e)}")
    
    except Exception as e:
        print(f"❌ Error general: {str(e)}")
        import traceback
        traceback.print_exc()

def consolidar_reportes_cftc_simple(ruta_directorio, nombre_archivo_entrada, nombre_archivo_salida):
    """
    Versión simple que adapta la lectura basada en el diagnóstico
    """
    try:
        ruta_completa_entrada = os.path.join(ruta_directorio, nombre_archivo_entrada)
        ruta_completa_salida = os.path.join(ruta_directorio, nombre_archivo_salida)
        
        print(f"🚀 INICIANDO CONSOLIDACIÓN SIMPLIFICADA")
        print(f"📁 Archivo: {ruta_completa_entrada}")
        
        if not os.path.exists(ruta_completa_entrada):
            print(f"❌ Archivo no encontrado")
            return None
        
        xls = pd.ExcelFile(ruta_completa_entrada)
        todos_los_datos = []
        
        print(f"\n📊 Procesando {len([h for h in xls.sheet_names if h.startswith('Dia_')])} hojas...")
        
        for hoja in xls.sheet_names:
            if hoja.startswith('Dia_'):
                try:
                    num_dia = hoja.split('_')[1]
                    
                    # PRIMERO: Intentar diferentes estrategias de lectura
                    df = None
                    
                    # Estrategia 1: Saltar 0 filas
                    try:
                        df_test = pd.read_excel(xls, sheet_name=hoja, skiprows=0)
                        if not df_test.empty and len(df_test.columns) > 1:
                            df = df_test
                            print(f"✅ {hoja}: Leído con skiprows=0 - {len(df)} registros")
                    except:
                        pass
                    
                    # Estrategia 2: Saltar 2 filas (la original)
                    if df is None:
                        try:
                            df_test = pd.read_excel(xls, sheet_name=hoja, skiprows=2)
                            if not df_test.empty and len(df_test.columns) > 1:
                                df = df_test
                                print(f"✅ {hoja}: Leído con skiprows=2 - {len(df)} registros")
                        except:
                            pass
                    
                    # Estrategia 3: Saltar 1 fila
                    if df is None:
                        try:
                            df_test = pd.read_excel(xls, sheet_name=hoja, skiprows=1)
                            if not df_test.empty and len(df_test.columns) > 1:
                                df = df_test
                                print(f"✅ {hoja}: Leído con skiprows=1 - {len(df)} registros")
                        except:
                            pass
                    
                    # Estrategia 4: Leer sin header
                    if df is None:
                        try:
                            df_test = pd.read_excel(xls, sheet_name=hoja, header=None)
                            if not df_test.empty and len(df_test.columns) > 1:
                                df = df_test
                                print(f"✅ {hoja}: Leído sin header - {len(df)} registros")
                        except:
                            pass
                    
                    if df is not None and not df.empty:
                        # Extraer fecha
                        try:
                            fecha_reporte = datetime(2025, 10, int(num_dia))
                        except:
                            fecha_reporte = datetime(2025, 10, 1)
                        
                        # Agregar columnas de identificación
                        df['Fecha_Reporte'] = fecha_reporte
                        df['Dia_Hoja'] = hoja
                        df['Dia_Numero'] = num_dia.zfill(2)
                        
                        todos_los_datos.append(df)
                    else:
                        print(f"❌ {hoja}: No se pudieron leer datos")
                        
                except Exception as e:
                    print(f"❌ Error en {hoja}: {str(e)}")
        
        if not todos_los_datos:
            print("\n❌ No se encontraron datos válidos")
            return None
        
        # Consolidar
        df_consolidado = pd.concat(todos_los_datos, ignore_index=True)
        
        # Guardar
        with pd.ExcelWriter(ruta_completa_salida, engine='openpyxl') as writer:
            df_consolidado.to_excel(writer, sheet_name='CONSOLIDADO', index=False)
        
        print(f"\n🎉 CONSOLIDACIÓN EXITOSA!")
        print(f"📊 Total registros: {len(df_consolidado)}")
        print(f"💾 Archivo guardado: {ruta_completa_salida}")
        
        return df_consolidado
        
    except Exception as e:
        print(f"❌ Error general: {str(e)}")
        return None
