import os
import pandas as pd
import gzip
import csv
from io import StringIO

def procesar_archivos_cftc_corregido():
    ruta_base = r"X:\REGULATORIO MERCADOS\FICHEROS DTCC PORTAL EU\DFA\2025\10. Octubre"
    
    patrones = [
        "CFTC_IR_SubmissionReport_*-en-6STB.csv.gz",
        "CFTC_FX_SubmissionReport_*-en-5STB.csv.gz", 
        "CFTC_EQ_SubmissionReport_*-en-4STB.csv.gz",
        "CFTC_CR_SubmissionReport_*-en-3STB.csv.gz"
    ]
    
    todos_datos = []
    
    for dia in range(1, 32):
        carpeta_dia = f"{dia:02d}.10"
        ruta_carpeta = os.path.join(ruta_base, carpeta_dia)
        
        if not os.path.exists(ruta_carpeta):
            continue
            
        print(f"Procesando: {carpeta_dia}")
        
        for patron in patrones:
            import glob
            ruta_patron = os.path.join(ruta_carpeta, patron)
            archivos = glob.glob(ruta_patron)
            
            for archivo in archivos:
                print(f"  → {os.path.basename(archivo)}")
                
                try:
                    # ENFOQUE 1: Leer como texto y analizar manualmente
                    with gzip.open(archivo, 'rt', encoding='utf-8') as f:
                        contenido = f.read()
                    
                    # Dividir en líneas y procesar
                    lineas = contenido.strip().split('\n')
                    
                    # Buscar encabezados válidos (líneas con múltiples campos)
                    datos_procesados = []
                    for linea in lineas:
                        # Intentar detectar el delimitador
                        if ';' in linea and linea.count(';') > 1:
                            campos = linea.split(';')
                        elif ',' in linea and linea.count(',') > 1:
                            campos = linea.split(',')
                        elif '|' in linea and linea.count('|') > 1:
                            campos = linea.split('|')
                        else:
                            campos = [linea]
                        
                        # Solo agregar líneas que tengan múltiples campos
                        if len(campos) > 1:
                            datos_procesados.append(campos)
                    
                    if len(datos_procesados) > 1:
                        # Crear DataFrame
                        df = pd.DataFrame(datos_procesados[1:], columns=datos_procesados[0])
                        df['Archivo_Origen'] = os.path.basename(archivo)
                        df['Fecha_Dia'] = f"2025-10-{dia:02d}"
                        todos_datos.append(df)
                        print(f"    ✓ Procesado: {len(df)} registros")
                    else:
                        print(f"    ⚠ No hay datos válidos en {archivo}")
                        
                except Exception as e:
                    print(f"    ✗ Error: {str(e)[:100]}...")
                    # Intentar enfoque alternativo
                    try:
                        df = procesar_con_pandas_directo(archivo)
                        if df is not None:
                            df['Archivo_Origen'] = os.path.basename(archivo)
                            df['Fecha_Dia'] = f"2025-10-{dia:02d}"
                            todos_datos.append(df)
                            print(f"    ✓ Recuperado con método alternativo: {len(df)} registros")
                    except:
                        print(f"    ✗ Método alternativo también falló")
    
    # Guardar resultados
    if todos_datos:
        df_combinado = pd.concat(todos_datos, ignore_index=True)
        nombre_excel = "CFTC_Combined_Corregido.xlsx"
        
        with pd.ExcelWriter(nombre_excel, engine='openpyxl') as writer:
            df_combinado.to_excel(writer, sheet_name='Datos_Combinados', index=False)
            
            # Crear hoja de resumen
            resumen = pd.DataFrame({
                'Total_Archivos': [len(todos_datos)],
                'Total_Registros': [len(df_combinado)],
                'Columnas': [', '.join(df_combinado.columns)]
            })
            resumen.to_excel(writer, sheet_name='Resumen', index=False)
        
        print(f"\n✅ PROCESO COMPLETADO")
        print(f"📊 Archivos procesados: {len(todos_datos)}")
        print(f"📈 Registros totales: {len(df_combinado)}")
        print(f"💾 Guardado como: {nombre_excel}")
        
        return df_combinado
    else:
        print("❌ No se pudieron procesar archivos")
        return None

def procesar_con_pandas_directo(archivo):
    """Enfoque alternativo para leer archivos problemáticos"""
    try:
        # Probar diferentes delimitadores
        for delimiter in [',', ';', '|', '\t']:
            try:
                with gzip.open(archivo, 'rt', encoding='utf-8') as f:
                    # Leer primeras líneas para diagnóstico
                    lineas = [f.readline() for _ in range(5)]
                    f.seek(0)  # Volver al inicio
                    
                    df = pd.read_csv(f, delimiter=delimiter, engine='python', 
                                   quoting=csv.QUOTE_MINIMAL, on_bad_lines='skip')
                    
                    if len(df.columns) > 1 and len(df) > 0:
                        return df
            except:
                continue
        
        # Último intento: leer todo como texto
        with gzip.open(archivo, 'rt', encoding='utf-8') as f:
            contenido = f.read()
        
        # Buscar patrones de datos
        lineas = contenido.split('\n')
        datos = []
        for linea in lineas:
            if any(x in linea for x in [',', ';', '|']) and len(linea) > 10:
                datos.append(linea)
        
        if datos:
            # Crear DataFrame manual
            df = pd.DataFrame([x.split(',') for x in datos if ',' in x])
            return df
            
    except Exception as e:
        print(f"    Método alternativo falló: {e}")
    
    return None
