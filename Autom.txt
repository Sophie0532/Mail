import pandas as pd
import os
import glob
import sys
from datetime import datetime

def encontrar_archivo_daily_mark(ruta_base):
    """
    Encuentra el archivo DailyMark más reciente basado en el patrón de fecha
    """
    patron = os.path.join(ruta_base, "*_DailyMark.xlsx")
    archivos = glob.glob(patron)
    
    if not archivos:
        print("❌ No se encontraron archivos DailyMark con el patrón de fecha")
        return None
    
    # Ordenar por fecha de modificación (más reciente primero)
    archivo_mas_reciente = max(archivos, key=os.path.getmtime)
    return archivo_mas_reciente

def obtener_fecha_analizada(df_daily):
    """
    Obtiene la fecha analizada del campo Communication date en DailyMark
    """
    try:
        print("\n🔍 Buscando columna de fecha en DailyMark...")
        print(f"   Columnas disponibles: {list(df_daily.columns)}")
        
        # Buscar la columna Communication date (puede tener variaciones)
        posibles_columnas_fecha = [
            'Communication date', 
            'Communication Date',
            'COMMUNICATION DATE',
            'communication date',
            'Date',
            'DATE',
            'Trade Date',
            'TradeDate',
            'Valuation Date',
            'ValuationDate'
        ]
        
        columna_encontrada = None
        for columna in posibles_columnas_fecha:
            if columna in df_daily.columns:
                columna_encontrada = columna
                break
        
        if columna_encontrada:
            print(f"   ✅ Columna encontrada: '{columna_encontrada}'")
            
            # Tomar el primer Communication date no nulo
            fecha_comunicacion = df_daily[columna_encontrada].dropna().iloc[0]
            print(f"   📅 Valor de fecha encontrado: {fecha_comunicacion}")
            print(f"   📊 Tipo de dato: {type(fecha_comunicacion)}")
            
            # Convertir a datetime
            if isinstance(fecha_comunicacion, str):
                # Intentar diferentes formatos de fecha
                formatos_fecha = ['%Y-%m-%d', '%d/%m/%Y', '%m/%d/%Y', '%Y.%m.%d', '%d-%m-%Y', '%m-%d-%Y']
                for formato in formatos_fecha:
                    try:
                        fecha = datetime.strptime(fecha_comunicacion, formato)
                        break
                    except ValueError:
                        continue
                else:
                    # Si ninguno funciona, usar pandas
                    fecha = pd.to_datetime(fecha_comunicacion)
            else:
                fecha = pd.to_datetime(fecha_comunicacion)
            
            print(f"   🔄 Fecha convertida: {fecha}")
            
            # Formatear como "Día 18 de Octubre analizado"
            dia = fecha.day
            mes = fecha.strftime('%B').lower()
            
            # Traducir meses al español
            meses_espanol = {
                'january': 'enero', 'february': 'febrero', 'march': 'marzo',
                'april': 'abril', 'may': 'mayo', 'june': 'junio',
                'july': 'julio', 'august': 'agosto', 'september': 'septiembre',
                'october': 'octubre', 'november': 'noviembre', 'deccember': 'diciembre'
            }
            
            mes_espanol = meses_espanol.get(mes, mes)
            
            resultado = f"Día {dia} de {mes_espanol.capitalize()} analizado"
            print(f"   ✅ Fecha formateada: {resultado}")
            return resultado
        
        else:
            print("   ❌ No se encontró ninguna columna de fecha conocida")
            # Mostrar todas las columnas disponibles para debugging
            print("   📋 Todas las columnas disponibles:")
            for i, columna in enumerate(df_daily.columns, 1):
                print(f"      {i:2d}. {columna}")
            return "Fecha no disponible - Columna no encontrada"
            
    except Exception as e:
        print(f"   ❌ Error al obtener fecha: {e}")
        import traceback
        print(f"   📝 Detalles del error: {traceback.format_exc()}")
        return f"Fecha no disponible - Error: {str(e)}"

def leer_csv_con_errores(archivo_phoenix):
    """
    Lee el archivo CSV manejando posibles errores de formato
    """
    try:
        # Primero intentamos leer normalmente
        return pd.read_csv(archivo_phoenix)
    except pd.errors.ParserError as e:
        print(f"⚠️  Error al leer CSV: {e}")
        print("🔧 Intentando métodos alternativos...")
        
        # Método 1: Especificar engine de Python
        try:
            print("   • Intentando con engine='python'...")
            return pd.read_csv(archivo_phoenix, engine='python')
        except:
            pass
        
        # Método 2: Leer con diferentes delimitadores
        try:
            print("   • Probando diferentes delimitadores...")
            for delimiter in [',', ';', '\t', '|']:
                try:
                    df = pd.read_csv(archivo_phoenix, delimiter=delimiter, engine='python')
                    print(f"   ✅ CSV leído con delimitador: '{delimiter}'")
                    return df
                except:
                    continue
        except:
            pass
        
        # Método 3: Leer sin encabezados y analizar estructura
        try:
            print("   • Analizando estructura del archivo...")
            with open(archivo_phoenix, 'r', encoding='utf-8') as f:
                lineas = f.readlines()
            
            print(f"   • El archivo tiene {len(lineas)} líneas")
            print(f"   • Primera línea: {lineas[0][:100]}...")
            print(f"   • Línea 104: {lineas[103] if len(lineas) > 103 else 'No existe'}")
            
            # Intentar leer saltando la línea problemática
            print("   • Intentando leer saltando líneas problemáticas...")
            return pd.read_csv(archivo_phoenix, error_bad_lines=False, warn_bad_lines=True, engine='python')
        except Exception as e:
            print(f"❌ No se pudo leer el archivo CSV: {e}")
            raise

        def analizar_valores_mtm(df, nombre_archivo, columna_id='USI'):
            """
            Analiza valores MTM: ceros, nulos y vacíos, y muestra los IDs correspondientes
            """
            print(f"   • {nombre_archivo}:")
            resultados = {}
            
            if 'MTM' not in df.columns:
                print(f"     - MTM: ❌ Columna no encontrada")
                return {
                    'ceros': 0,
                    'nulos': 0,
                    'vacios': 0,
                    'total_ceros': 0,
                    'ids_ceros': [],
                    'ids_vacios': []
                }
            
            # Contar valores cero (0)
            ceros = (df['MTM'] == 0).sum()
            
            # Contar valores nulos
            nulos = df['MTM'].isna().sum()
            
            # Contar strings vacíos
            vacios = 0
            if df['MTM'].dtype == 'object':
                vacios = (df['MTM'].astype(str).str.strip() == '').sum()
            
            # Obtener IDs con valores cero
            ids_ceros = []
            if ceros > 0 and columna_id in df.columns:
                ids_ceros = df[df['MTM'] == 0][columna_id].dropna().astype(str).tolist()
            
            # Obtener IDs con valores vacíos/nulos
            ids_vacios = []
            if (nulos + vacios) > 0 and columna_id in df.columns:
                # IDs con valores nulos
                if nulos > 0:
                    ids_nulos = df[df['MTM'].isna()][columna_id].dropna().astype(str).tolist()
                    ids_vacios.extend(ids_nulos)
                
                # IDs con strings vacíos
                if vacios > 0:
                    ids_str_vacios = df[df['MTM'].astype(str).str.strip() == ''][columna_id].dropna().astype(str).tolist()
                    ids_vacios.extend(ids_str_vacios)
            
            resultados = {
                'ceros': ceros,
                'nulos': nulos,
                'vacios': vacios,
                'total_ceros': ceros,
                'total_vacios': nulos + vacios,
                'ids_ceros': ids_ceros,
                'ids_vacios': ids_vacios
            }
            
            print(f"     - MTM - Valores cero: {ceros}")
            print(f"     - MTM - Valores nulos: {nulos}")
            print(f"     - MTM - Valores vacíos: {vacios}")
            
            return resultados
        
        # Analizar DailyMark
        print("\n   📋 DailyMark:")
        resultados_mtm_daily = analizar_valores_mtm(df_daily, "DailyMark", 'USI')
        
        # Analizar Phoenix
        print("\n   📋 Phoenix:")
