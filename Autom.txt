import pandas as pd
import os
import glob
import sys
from datetime import datetime

def encontrar_archivo_daily_mark(ruta_base):
    """
    Encuentra el archivo DailyMark más reciente basado en el patrón de fecha
    """
    patron = os.path.join(ruta_base, "*_DailyMark.xlsx")
    archivos = glob.glob(patron)
    
    if not archivos:
        print("❌ No se encontraron archivos DailyMark con el patrón de fecha")
        return None
    
    # Ordenar por fecha de modificación (más reciente primero)
    archivo_mas_reciente = max(archivos, key=os.path.getmtime)
    return archivo_mas_reciente

def leer_csv_con_errores(archivo_phoenix):
    """
    Lee el archivo CSV manejando posibles errores de formato
    """
    try:
        # Primero intentamos leer normalmente
        return pd.read_csv(archivo_phoenix)
    except pd.errors.ParserError as e:
        print(f"⚠️  Error al leer CSV: {e}")
        print("🔧 Intentando métodos alternativos...")
        
        # Método 1: Especificar engine de Python
        try:
            print("   • Intentando con engine='python'...")
            return pd.read_csv(archivo_phoenix, engine='python')
        except:
            pass
        
        # Método 2: Leer con diferentes delimitadores
        try:
            print("   • Probando diferentes delimitadores...")
            for delimiter in [',', ';', '\t', '|']:
                try:
                    df = pd.read_csv(archivo_phoenix, delimiter=delimiter, engine='python')
                    print(f"   ✅ CSV leído con delimitador: '{delimiter}'")
                    return df
                except:
                    continue
        except:
            pass
        
        # Método 3: Leer sin encabezados y analizar estructura
        try:
            print("   • Analizando estructura del archivo...")
            with open(archivo_phoenix, 'r', encoding='utf-8') as f:
                lineas = f.readlines()
            
            print(f"   • El archivo tiene {len(lineas)} líneas")
            print(f"   • Primera línea: {lineas[0][:100]}...")
            print(f"   • Línea 104: {lineas[103] if len(lineas) > 103 else 'No existe'}")
            
            # Intentar leer saltando la línea problemática
            print("   • Intentando leer saltando líneas problemáticas...")
            return pd.read_csv(archivo_phoenix, error_bad_lines=False, warn_bad_lines=True, engine='python')
        except Exception as e:
            print(f"❌ No se pudo leer el archivo CSV: {e}")
            raise

def comparar_archivos():

            return

        print("=" * 60)

        print(f"📁 Archivo Phoenix: {os.path.basename(archivo_phoenix)}")
        print(f"📂 Ruta: {ruta_base}")
        print("=" * 60)
        
        # Leer archivos
        print("📖 Leyendo archivos...")
        
        # Leer Excel (DailyMark)
        df_daily = pd.read_excel(archivo_daily_mark)
        print(f"✅ DailyMark leído: {len(df_daily)} filas, {len(df_daily.columns)} columnas")
        
        # Leer CSV (Phoenix) con manejo de errores
        df_phoenix = leer_csv_con_errores(archivo_phoenix)
        print(f"✅ Phoenix leído: {len(df_phoenix)} filas, {len(df_phoenix.columns)} columnas")
        
        # Mostrar información de las columnas
        print("\n📊 COLUMNAS ENCONTRADAS:")
        print(f"   • DailyMark: {list(df_daily.columns)}")
        print(f"   • Phoenix: {list(df_phoenix.columns)}")
        print()
        
        # 1. CONTEO DE FILAS TOTALES
        print("1. 📊 CONTEO DE FILAS TOTALES")
        print(f"   • DailyMark: {len(df_daily):,} filas")
        print(f"   • Phoenix: {len(df_phoenix):,} filas")
        print()
        
        # 2. ANÁLISIS DE VALORES NULL Y VACÍOS
        print("2. 🔍 ANÁLISIS DE VALORES NULL Y VACÍOS")
        
        def analizar_nulos_y_vacios(df, nombre_archivo, columna_id='USI'):
            """
            Analiza valores nulos y vacíos en columnas importantes y muestra los IDs correspondientes
            """
            print(f"   • {nombre_archivo}:")
            resultados = {}
            
            # Columnas a analizar
            columnas_analizar = ['USI', 'UTI', 'MTM', 'Counterparty', 'TradeDate', 'ProductType']
            
            for columna in columnas_analizar:
                if columna in df.columns:
                    # Contar nulos
                    nulos = df[columna].isna().sum()
                    
                    # Contar strings vacíos (solo para columnas de texto)
                    vacios = 0
                    if df[columna].dtype == 'object':
                        vacios = (df[columna].astype(str).str.strip() == '').sum()
                    
                    total_problemas = nulos + vacios
                    
                    # Obtener IDs con problemas
                    ids_problemas = []
                    if total_problemas > 0 and columna_id in df.columns:
                        # IDs con valores nulos
                        if nulos > 0:
                            ids_nulos = df[df[columna].isna()][columna_id].dropna().astype(str).tolist()
                            ids_problemas.extend(ids_nulos)
                        
                        # IDs con strings vacíos
                        if vacios > 0:
                            ids_vacios = df[df[columna].astype(str).str.strip() == ''][columna_id].dropna().astype(str).tolist()
                            ids_problemas.extend(ids_vacios)
                    
                    resultados[columna] = {
                        'nulos': nulos, 
                        'vacios': vacios,
                        'total_problemas': total_problemas,
                        'ids_problemas': ids_problemas
                    }
                    
                    if total_problemas > 0:
                        print(f"     - {columna}: {total_problemas} problemas ({nulos} nulos, {vacios} vacíos)")
                        if ids_problemas:
                            print(f"       {columna_id}s afectados: {ids_problemas}")
                    else:
                        print(f"     - {columna}: 0 problemas")
                else:
                    print(f"     - {columna}: ❌ Columna no encontrada")
                    resultados[columna] = {
                        'nulos': 0, 
                        'vacios': 0,
                        'total_problemas': 0,
                        'ids_problemas': []
                    }
            
            return resultados
        
        # Analizar DailyMark
        print("\n   📋 DailyMark:")
        resultados_daily = analizar_nulos_y_vacios(df_daily, "DailyMark", 'USI')
        
        # Analizar Phoenix
        print("\n   📋 Phoenix:")
        resultados_phoenix = analizar_nulos_y_vacios(df_phoenix, "Phoenix", 'UTI')
        print()
        
        # Obtener USI/UTI para valores vacíos MTM
        def obtener_ids_mtm_vacios(df, columna_mtm='MTM', columna_id='USI'):
            """Obtiene los IDs correspondientes a valores MTM vacíos"""
            if columna_mtm not in df.columns or columna_id not in df.columns:
                return []
            
            # Filtrar filas donde MTM está vacío o es nulo
            mask = (df[columna_mtm].isna()) | (df[columna_mtm].astype(str).str.strip() == '')
            ids_vacios = df[mask][columna_id].dropna().astype(str).tolist()
            return ids_vacios
        
        # Obtener USI/UTI para MTM vacíos
        usi_mtm_vacios_daily = obtener_ids_mtm_vacios(df_daily, 'MTM', 'USI')
        uti_mtm_vacios_phoenix = obtener_ids_mtm_vacios(df_phoenix, 'MTM', 'UTI')
        
        # RESUMEN FINAL
        print("=" * 60)
        print("📈 RESUMEN FINAL")
        print("=" * 60)
        print(f"📊 Total filas:")
        print(f"   • DailyMark: {len(df_daily):,}")
        print(f"   • Phoenix: {len(df_phoenix):,}")
        print()
        
        # Mostrar valores cero y vacíos MTM
        mtm_cero_daily = resultados_daily.get('MTM', {}).get('nulos', 0) + resultados_daily.get('MTM', {}).get('vacios', 0)
        mtm_cero_phoenix = resultados_phoenix.get('MTM', {}).get('nulos', 0) + resultados_phoenix.get('MTM', {}).get('vacios', 0)
        
        print(f"🔍 VALORES CERO EN CAMPO MTM:")
        print(f"   • DailyMark - Valores cero: {mtm_cero_daily}")
        print(f"   • Phoenix - Valores cero: {mtm_cero_phoenix}")
        print()
        
        print(f"📝 VALORES VACÍOS MTM:")
        print(f"   • DailyMark - Valores vacíos: {len(usi_mtm_vacios_daily)}")
        if usi_mtm_vacios_daily:
            print(f"     USIs correspondientes: {usi_mtm_vacios_daily}")
        
        print(f"   • Phoenix - Valores vacíos: {len(uti_mtm_vacios_phoenix)}")
        if uti_mtm_vacios_phoenix:
            print(f"     UTIs correspondientes: {uti_mtm_vacios_phoenix}")
        
        print("=" * 60)
        
    except Exception as e:
        print(f"❌ Error: {str(e)}")
        print("💡 Posibles soluciones:")
        print("   - Verifica que tengas acceso a la ruta de red")

        print("   - Verifica el formato del archivo CSV (puede tener comas adicionales)")
